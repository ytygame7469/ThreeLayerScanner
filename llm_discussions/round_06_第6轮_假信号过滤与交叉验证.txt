### 1. 成分股位置分布过滤
```python
def position_distribution_filter(sector_stocks, lookback_days=60):
    """
    计算成分股位置分布，避免平均化陷阱
    """
    positions = []
    for stock in sector_stocks:
        high = max(stock.close[-lookback_days:])
        low = min(stock.close[-lookback_days:])
        current = stock.close[-1]
        position = (current - low) / (high - low) * 100  # 0-100位置
        positions.append(position)
    
    # 关键指标：中位数和分布形态
    median_position = np.median(positions)
    q25, q75 = np.percentile(positions, [25, 75])
    skewness = stats.skew(positions)
    
    # 过滤条件：避免极端位置
    if median_position < 20 or median_position > 80:
        return False, "板块位置极端，非理想入场点"
    
    # 分布形态判断：正偏态(右偏)表示龙头股位置高
    if skewness > 1.5:
        return False, "龙头股已大幅拉升，跟风股滞后"
    
    return True, "位置分布合理"
```

### 2. 龙头与跟风乖离率过滤
```python
def leader_follower_divergence(leader_stock, follower_stocks, period=20):
    """
    龙头股与跟风股乖离率检测
    """
    leader_ma = leader_stock.close[-period:].mean()
    
    divergences = []
    for follower in follower_stocks:
        follower_ma = follower.close[-period:].mean()
        divergence = (leader_ma - follower_ma) / follower_ma * 100
        divergences.append(divergence)
    
    avg_divergence = np.mean(divergences)
    
    # 阈值：乖离率超过15%警示，超过25%拒绝
    if abs(avg_divergence) > 25:
        return False, f"龙头跟风乖离率过大: {avg_divergence:.1f}%"
    elif abs(avg_divergence) > 15:
        return True, f"警示：乖离率偏高: {avg_divergence:.1f}%"
    
    return True, "乖离率正常"
```

### 3. 历史准确率回测过滤
```python
def historical_accuracy_filter(signal_pattern, history_data, threshold=0.6):
    """
    回测历史信号准确率
    """
    correct_signals = 0
    total_signals = 0
    
    for i in range(len(history_data) - 20):
        if detect_signal(history_data[i:i+20]) == signal_pattern:
            total_signals += 1
            # 检查未来10日收益
            future_return = (history_data[i+30] - history_data[i+20]) / history_data[i+20]
            if future_return > 0.05:  # 5%止盈标准
                correct_signals += 1
    
    accuracy = correct_signals / total_signals if total_signals > 0 else 0
    
    return accuracy > threshold, f"历史准确率: {accuracy:.2%}"
```

```python
def multi_timeframe_validation(daily_data, weekly_data):
    """
    日线和周线级别交叉验证
    """
    # 日线信号
    daily_signal = calculate_macd_signal(daily_data)
    daily_rsi = calculate_rsi(daily_data.close, 14)
    
    # 周线信号
    weekly_signal = calculate_macd_signal(weekly_data)
    weekly_trend = "up" if weekly_data.close[-1] > weekly_data.ma(20)[-1] else "down"
    
    # 验证条件
    score = 0
    
    if daily_signal == "buy" and weekly_signal == "buy":
        score += 3
    elif daily_signal == "buy" and weekly_trend == "up":
        score += 2
    elif daily_signal == "buy":
        score += 1
    
    if 30 < daily_rsi < 70:
        score += 1
    
    # 阈值：至少3分才通过
    return score >= 3, f"多周期验证分数: {score}/4"
```

### 5. 量能验证过滤
```python
def volume_validation(stock_data, period=5):
    """
    量能验证：放量可信度判断
    """
    current_volume = stock_data.volume[-1]
    avg_volume = stock_data.volume[-period-1:-1].mean()
    volume_ratio = current_volume / avg_volume
    
    # 价格变动
    price_change = (stock_data.close[-1] - stock_data.close[-2]) / stock_data.close[-2]
    
    # 量价关系判断
    if volume_ratio > 2.0 and price_change > 0.03:
        return True, "放量上涨，可信度高"
    elif volume_ratio > 2.0 and price_change < -0.03:
        return False, "放量下跌，风险信号"
    elif volume_ratio < 0.5 and abs(price_change) < 0.01:
        return False, "缩量横盘，方向不明"
    elif volume_ratio < 0.5 and price_change > 0.02:
        return True, "缩量上涨，筹码锁定好"
    elif 0.8 < volume_ratio < 1.2:
        return True, "正常量能，信号有效"
    
    return True, "量能正常"
```

### 6. 综合过滤算法伪代码

```python
class SectorRotationFilter:
    def __init__(self):
        self.thresholds = {
            'position_median_min': 20,
            'position_median_max': 80,
            'skewness_max': 1.5,
            'divergence_warning': 15,
            'divergence_reject': 25,
            'historical_accuracy_min': 0.6,
            'timeframe_score_min': 3,
            'volume_ratio_high': 2.0,
            'volume_ratio_low': 0.5
        }
    
    def comprehensive_filter(self, sector_data):
        """
        综合多层过滤
        """
        filter_results = {}
        total_score = 0
        
        # 1. 位置分布过滤
        passed, msg = position_distribution_filter(sector_data.stocks)
        filter_results['position'] = {'passed': passed, 'message': msg}
        if passed:
            total_score += 2
        
        # 2. 龙头跟风乖离率
        passed, msg = leader_follower_divergence(
            sector_data.leader, sector_data.followers
        )
        filter_results['divergence'] = {'passed': passed, 'message': msg}
        if passed:
            total_score += 2
        
        # 3. 历史准确率回测
        passed, msg = historical_accuracy_filter(
            sector_data.current_signal, sector_data.history
        )
        filter_results['historical'] = {'passed': passed, 'message': msg}
        if passed:
            total_score += 1
        
        # 4. 多周期验证
        passed, msg = multi_timeframe_validation(
            sector_data.daily, sector_data.weekly
        )
        filter_results['timeframe'] = {'passed': passed, 'message': msg}
        if passed:
            total_score += 2
        
        # 5. 量能验证
        passed, msg = volume_validation(sector_data.leader_data)
        filter_results['volume'] = {'passed': passed, 'message': msg}
        if passed:
            total_score += 1
        
        # 最终决策
        decision = self.make_decision(total_score, filter_results)
        
        return {
            'decision': decision,
            'score': total_score,
            'details': filter_results
        }
    
    def make_decision(self, score, results):
        """
        决策逻辑
        """
        # 硬性否决条件
        if not results['position']['passed']:
            return "REJECT: 板块位置异常"
        
        if not results['divergence']['passed']:
            return "REJECT: 龙头跟风乖离率过大"
        
        # 评分决策
        if score >= 7:
            return "STRONG_BUY: 强买入信号"
        elif score >= 5:
            return "BUY: 买入信号"
        elif score >= 3:
            return "WATCH: 观察信号"
        else:
            return "AVOID: 规避信号"
```

| 过滤层 | 参数 | 阈值 | 说明 |
|--------|------|------|------|
| 位置分布 | 中位数位置 | 20-80 | 避免极端位置 |
| 位置分布 | 偏度 | <1.5 | 防止龙头过度拉升 |
| 乖离率 | 警示线 | 15% | 提示风险 |
| 乖离率 | 拒绝线 | 25% | 直接否决 |

这套方案通过多层过滤避免了AI算力上游的误判问题，重点关注分布形态而非平均值，能有效识别真假板块轮动信号。
